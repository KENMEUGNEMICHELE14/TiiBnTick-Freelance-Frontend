   590→                  <div>

   647→                        <Lock className="w-5 h-5 text-gray-400 shrink-0" />

   649→                          {showPassword ? clientData.password : '********'}
   650→                        </span>
   651→                        <button
   652→                          type="button"
   653→                          onClick={() => setShowPassword(!showPassword)}
   654→                          className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-orange-500 transition-colors"
   655→                          aria-label={showPassword ? "Masquer" : "Afficher"}
   656→                        >
   657→                          {showPassword ? <Eye size={18} /> : <EyeOff size={18} />}
   658→                        </button>
   659→                      </div>
   660→                    )}
   661→                  </div>
   662→                </div>
   663→
   664→                {isEditing && (
   665→                  <div className="flex gap-3 pt-4">
   666→                    <Button
   667→                      onClick={handleSave}
   668→                      disabled={isLoading}
   669→                      className="flex-1 bg-gradient-to-r from-orange-500 to-amber-500 hover:from-orange-600 hover:to-amber-600 text-white"

   671→                      <span className="flex items-center justify-center gap-2 pointer-events-none">

   675→                          <CheckCircle2 key="check" className="w-4 h-4" />
   676→                        )}
   677→                        <span className="font-medium">Enregistrer</span>
   678→                      </span>
   679→                    </Button>
   680→                    <Button
   681→                      onClick={handleCancel}
   682→                      variant="outline"
   683→                      className="flex-1"
   684→                    >
   685→                      Annuler
   686→                    </Button>
   687→                  </div>
   688→                )}
   689→              </CardContent>
   690→            </Card>
   691→
   692→            {/* Account Actions Card */}
   693→            <Card>
   694→              <CardHeader>
   695→                <CardTitle className="flex items-center gap-2">
   696→                  <Settings className="w-5 h-5 text-orange-600" />
   697→                  Actions du compte
   698→                </CardTitle>
   699→              </CardHeader>
   700→              <CardContent className="space-y-4">
   701→                <Button
   702→                  onClick={() => setIsEditing(true)}
   703→                  variant="outline"
   704→                  className="w-full justify-start h-12 border-2 border-orange-200 hover:bg-orange-50 hover:border-orange-300 text-orange-700"

   706→                  <Edit className="w-5 h-5 mr-3" />
   707→                  <span className="font-medium">Modifier mon compte</span>
   708→                </Button>
   709→
   710→                <Button
   711→                  onClick={handleLogout}
   712→                  variant="outline"
   713→                  className="w-full justify-start h-12 border-2 border-blue-200 hover:bg-blue-50 hover:border-blue-300 text-blue-700"
   714→                >
   715→                  <LogOut className="w-5 h-5 mr-3" />
   716→                  <span className="font-medium">Déconnexion</span>
   717→                </Button>
   718→
   719→                <Button
   720→                  onClick={() => setShowDeleteDialog(true)}
   721→                  variant="outline"
   722→                  className="w-full justify-start h-12 border-2 border-orange-200 hover:bg-orange-50 hover:border-orange-300 text-orange-600"
   723→                >
   724→                  <Trash2 className="w-5 h-5 mr-3" />

   726→                </Button>
   727→              </CardContent>
   728→            </Card>
   729→
   730→            {/* Delete Confirmation Dialog */}
   731→            {showDeleteDialog && (
   732→              <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50 p-4">
   733→                <Card className="max-w-md w-full border-2 border-orange-200">
   734→                  <CardHeader>
   735→                    <div className="flex items-center gap-3">
   736→                      <div className="w-12 h-12 bg-orange-100 rounded-full flex items-center justify-center">
   737→                        <AlertTriangle className="w-6 h-6 text-orange-600" />
   738→                      </div>
   739→                      <div>
   740→                        <CardTitle className="text-xl text-orange-900">Supprimer le compte ?</CardTitle>
   741→                        <p className="text-sm text-orange-700 mt-1">Cette action est irréversible</p>
   742→                      </div>
   743→                    </div>
   744→                  </CardHeader>
   745→                  <CardContent>
   746→                    <p className="text-gray-600 mb-6">
   747→                      Êtes-vous sûr de vouloir supprimer votre compte ? Cette action est irréversible et toutes vos données seront définitivement perdues.
   748→                    </p>
   749→                    <div className="flex gap-3">
   750→                      <Button
   751→                        onClick={() => setShowDeleteDialog(false)}
   752→                        variant="outline"
   753→                        className="flex-1"
   754→                      >
   755→                        Non
   756→                      </Button>
   757→                      <Button
   758→                        onClick={handleDeleteAccount}
   759→                        disabled={isLoading}
   760→                        className="flex-1 bg-gradient-to-r from-orange-600 to-red-600 hover:from-orange-700 hover:to-red-700 text-white"
   761→                      >
   762→                        {isLoading ? (
   763→                          <Loader2 className="w-4 h-4 animate-spin mr-2" />
   764→                        ) : (
   765→                          <Trash2 className="w-4 h-4 mr-2" />
   766→                        )}
   767→                        Oui
   768→                      </Button>
   769→                    </div>
   770→                  </CardContent>
   771→                </Card>
   772→              </div>
   773→            )}
   774→          </div>
   775→        )}
   776→      </main>
   777→
   778→      {/* Desktop Footer */}
   779→      <footer className="hidden md:block py-6 bg-white border-t border-gray-100 mt-auto">
   780→        <div className="max-w-7xl mx-auto px-4 text-center text-sm text-gray-500">
   781→          <p>© 2025 TiiBnTick - Espace Go • Disponible 24h/24</p>
   782→        </div>

   784→    </div>
   785→  )
