   134→// Deterministic number formatter (avoids hydration mismatch between server and client)
   135→function fmt(n: number | undefined): string {
   136→  if (n == null) return '0'
   137→  return n.toString().replace(/\B(?=(\d{3})+(?!\d))/g, '\u00A0')
   138→}
