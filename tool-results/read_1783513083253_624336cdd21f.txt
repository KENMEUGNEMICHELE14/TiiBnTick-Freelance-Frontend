   104→// Deterministic number formatter (avoids hydration mismatch between server and client)
   105→function fmt(n: number | undefined): string {
   106→  if (n == null) return '0'
   107→  return n.toString().replace(/\B(?=(\d{3})+(?!\d))/g, '\u00A0')
   108→}
