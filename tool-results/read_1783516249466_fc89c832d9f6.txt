    31→// Inline GeoJSON Feature type (avoid importing 'geojson' package — crashes Turbopack)

    48→// Deterministic number formatter
    49→function fmt(n: number | undefined): string {
    50→  if (n == null) return '0'
    51→  return n.toString().replace(/\B(?=(\d{3})+(?!\d))/g, '\u00A0')
    52→}

   136→      : null
   137→
   138→  const deliveryCoords: [number, number] | null =
   139→    announcement?.deliveryAddress?.latitude && announcement?.deliveryAddress?.longitude
   140→      ? [announcement.deliveryAddress.latitude, announcement.deliveryAddress.longitude]
   141→      : null

   144→    pickupCoords && deliveryCoords

   490→            initial={{ y: '100%' }}
   491→            animate={{ y: 0 }}
   492→            exit={{ y: '100%' }}
   493→            transition={{ type: 'spring', stiffness: 300, damping: 30 }}
   494→            className="absolute inset-x-0 bottom-0 z-40 bg-white rounded-t-3xl shadow-[0_-4px_30px_rgba(0,0,0,0.15)] flex flex-col"
   495→            style={{ height: '75vh' }}
   496→          >
   497→            {/* Chat header */}
   498→            <div className="flex items-center justify-between px-4 py-3 border-b border-gray-100 flex-shrink-0">
   499→              <div className="flex items-center gap-3">
   500→                <div className="w-9 h-9 rounded-full bg-gradient-to-br from-orange-400 to-amber-500 flex items-center justify-center text-white font-bold text-sm">
   501→                  {announcement?.shipperFirstName?.charAt(0) || '?'}
   502→                </div>
   503→                <div>
   504→                  <p className="text-sm font-bold text-gray-900">{announcement?.shipperFirstName} {announcement?.shipperLastName}</p>
   505→                  <p className="text-[10px] text-green-500 flex items-center gap-1">
   506→                    <CircleDot className="w-2.5 h-2.5" /> En ligne
   507→                  </p>
   508→                </div>
   509→              </div>
   510→              <Button
   511→                variant="ghost"
   512→                size="icon"
   513→                className="rounded-full w-8 h-8"
   514→                onClick={() => setChatOpen(false)}
   515→              >
   516→                <X className="w-5 h-5 text-gray-500" />
   517→              </Button>
   518→            </div>
   519→
   520→            {/* Chat messages */}
   521→            <div className="flex-1 overflow-y-auto p-4 space-y-3">
   522→              {chatMessages.map((msg, i) => (
   523→                <div
   524→                  key={i}
   525→                  className={`flex ${msg.from === 'me' ? 'justify-end' : 'justify-start'}`}
   526→                >
   527→                  <div
   528→                    className={`max-w-[80%] px-4 py-2.5 rounded-2xl text-sm leading-relaxed ${
   529→                      msg.from === 'me'
   530→                        ? 'bg-orange-500 text-white rounded-br-md'
   531→                        : 'bg-gray-100 text-gray-800 rounded-bl-md'
   532→                    }`}
   533→                  >
   534→                    <p>{msg.text}</p>
   535→                    <p className={`text-[10px] mt-1 ${msg.from === 'me' ? 'text-orange-200' : 'text-gray-400'}`}>
   536→                      {msg.time}
   537→                    </p>
   538→                  </div>
   539→                </div>
   540→              ))}
   541→              <div ref={chatEndRef} />
   542→            </div>
   543→
   544→            {/* Chat input */}
   545→            <div className="px-4 py-3 border-t border-gray-100 flex items-center gap-2 flex-shrink-0">
   546→              <Input
   547→                value={chatInput}
   548→                onChange={(e) => setChatInput(e.target.value)}
   549→                onKeyDown={(e) => e.key === 'Enter' && handleSendChat()}
   550→                placeholder="Écrire un message..."
   551→                className="flex-1 rounded-full border-gray-200 focus:border-orange-400"
   552→              />
   553→              <Button
   554→                size="icon"
   555→                onClick={handleSendChat}
   556→                className="rounded-full w-10 h-10 bg-gradient-to-r from-orange-500 to-amber-500 hover:from-orange-600 hover:to-amber-600"
   557→              >
   558→                <Send className="w-4 h-4 text-white" />
   559→              </Button>
   560→            </div>
   561→          </motion.div>
   562→        )}
   563→      </AnimatePresence>
   564→
   565→      {/* ── No announcement fallback ── */}
   566→      {!loading && !announcement && (
   567→        <div className="absolute inset-0 flex items-center justify-center z-10 bg-white/80 backdrop-blur-sm">
   568→          <div className="text-center p-6">
   569→            <Package className="w-16 h-16 text-gray-300 mx-auto mb-4" />
   570→            <h3 className="text-lg font-bold text-gray-700 mb-2">Aucune livraison sélectionnée</h3>
   571→            <p className="text-sm text-gray-500 mb-4">Accédez à cette page depuis une annonce.</p>
   572→            <Button onClick={() => router.back()} variant="outline" className="rounded-full">
   573→              <ArrowLeft className="w-4 h-4 mr-2" /> Retour
   574→            </Button>
   575→          </div>
   576→        </div>
   577→      )}
   578→    </div>
   579→  )
   580→}
