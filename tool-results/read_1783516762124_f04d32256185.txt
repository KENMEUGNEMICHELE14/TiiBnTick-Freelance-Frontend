    45→// Deterministic number formatter
    46→function fmt(n: number | undefined): string {
    47→  if (n == null) return '0'
    48→  return n.toString().replace(/\B(?=(\d{3})+(?!\d))/g, '\u00A0')
    49→}

    91→            data.pickupAddress.longitude,

   131→      : null
   132→

   134→    announcement?.deliveryAddress?.latitude && announcement?.deliveryAddress?.longitude
   135→      ? [announcement.deliveryAddress.latitude, announcement.deliveryAddress.longitude]

   137→

   485→            initial={{ y: '100%' }}
   486→            animate={{ y: 0 }}
   487→            exit={{ y: '100%' }}
   488→            transition={{ type: 'spring', stiffness: 300, damping: 30 }}
   489→            className="absolute inset-x-0 bottom-0 z-40 bg-white rounded-t-3xl shadow-[0_-4px_30px_rgba(0,0,0,0.15)] flex flex-col"
   490→            style={{ height: '75vh' }}
   491→          >
   492→            {/* Chat header */}
   493→            <div className="flex items-center justify-between px-4 py-3 border-b border-gray-100 flex-shrink-0">
   494→              <div className="flex items-center gap-3">
   495→                <div className="w-9 h-9 rounded-full bg-gradient-to-br from-orange-400 to-amber-500 flex items-center justify-center text-white font-bold text-sm">
   496→                  {announcement?.shipperFirstName?.charAt(0) || '?'}
   497→                </div>
   498→                <div>
   499→                  <p className="text-sm font-bold text-gray-900">{announcement?.shipperFirstName} {announcement?.shipperLastName}</p>
   500→                  <p className="text-[10px] text-green-500 flex items-center gap-1">
   501→                    <CircleDot className="w-2.5 h-2.5" /> En ligne
   502→                  </p>
   503→                </div>
   504→              </div>
   505→              <Button
   506→                variant="ghost"
   507→                size="icon"
   508→                className="rounded-full w-8 h-8"
   509→                onClick={() => setChatOpen(false)}
   510→              >
   511→                <X className="w-5 h-5 text-gray-500" />
   512→              </Button>
   513→            </div>
   514→
   515→            {/* Chat messages */}
   516→            <div className="flex-1 overflow-y-auto p-4 space-y-3">
   517→              {chatMessages.map((msg, i) => (
   518→                <div
   519→                  key={i}
   520→                  className={`flex ${msg.from === 'me' ? 'justify-end' : 'justify-start'}`}
   521→                >
   522→                  <div
   523→                    className={`max-w-[80%] px-4 py-2.5 rounded-2xl text-sm leading-relaxed ${
   524→                      msg.from === 'me'
   525→                        ? 'bg-orange-500 text-white rounded-br-md'
   526→                        : 'bg-gray-100 text-gray-800 rounded-bl-md'
   527→                    }`}
   528→                  >
   529→                    <p>{msg.text}</p>
   530→                    <p className={`text-[10px] mt-1 ${msg.from === 'me' ? 'text-orange-200' : 'text-gray-400'}`}>
   531→                      {msg.time}
   532→                    </p>
   533→                  </div>
   534→                </div>
   535→              ))}
   536→              <div ref={chatEndRef} />
   537→            </div>
   538→
   539→            {/* Chat input */}
   540→            <div className="px-4 py-3 border-t border-gray-100 flex items-center gap-2 flex-shrink-0">
   541→              <Input
   542→                value={chatInput}
   543→                onChange={(e) => setChatInput(e.target.value)}
   544→                onKeyDown={(e) => e.key === 'Enter' && handleSendChat()}
   545→                placeholder="Écrire un message..."
   546→                className="flex-1 rounded-full border-gray-200 focus:border-orange-400"
   547→              />
   548→              <Button
   549→                size="icon"
   550→                onClick={handleSendChat}
   551→                className="rounded-full w-10 h-10 bg-gradient-to-r from-orange-500 to-amber-500 hover:from-orange-600 hover:to-amber-600"
   552→              >
   553→                <Send className="w-4 h-4 text-white" />
   554→              </Button>
   555→            </div>
   556→          </motion.div>
   557→        )}
   558→      </AnimatePresence>
   559→
   560→      {/* ── No announcement fallback ── */}
   561→      {!loading && !announcement && (
   562→        <div className="absolute inset-0 flex items-center justify-center z-10 bg-white/80 backdrop-blur-sm">
   563→          <div className="text-center p-6">
   564→            <Package className="w-16 h-16 text-gray-300 mx-auto mb-4" />
   565→            <h3 className="text-lg font-bold text-gray-700 mb-2">Aucune livraison sélectionnée</h3>
   566→            <p className="text-sm text-gray-500 mb-4">Accédez à cette page depuis une annonce.</p>
   567→            <Button onClick={() => router.back()} variant="outline" className="rounded-full">
   568→              <ArrowLeft className="w-4 h-4 mr-2" /> Retour
   569→            </Button>
   570→          </div>
   571→        </div>
   572→      )}
   573→    </div>
   574→  )
   575→}
