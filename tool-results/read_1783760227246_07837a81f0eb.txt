   550→            </CardContent>
   551→          </Card>
   552→
   553→          {/* Info sécurité */}
   554→          <Card className="border-orange-100 bg-orange-50/50">
   555→            <CardContent className="p-4">
   556→              <div className="flex items-start gap-3">
   557→                <Shield className="w-5 h-5 text-orange-500 mt-0.5 flex-shrink-0" />
   558→                <div>

   561→                    Vos fonds sont protégés. Les recharges sont traitées sous 24h ouvrables.
   562→                    Le montant minimum de recharge est de 500 FCFA.
   563→                  </p>
   564→                </div>
   565→              </div>
   566→            </CardContent>
   567→          </Card>
   568→        </div>
   569→      </main>
   570→
   571→      {/* Dialog Recharge */}

   574→          <DialogHeader>
   575→            <DialogTitle className="flex items-center gap-2">

   577→              Recharger votre compte

   592→                  <Smartphone className="w-6 h-6 text-orange-500" />

   594→                    <p className="text-sm font-semibold text-gray-900">Mobile Money</p>

   607→            </div>
   608→
   609→            {/* Solde actuel */}
   610→            <div className="flex items-center justify-between text-sm">
   611→              <span className="text-gray-500">Solde actuel</span>

   613→            </div>
   614→
   615→            {/* Montant */}
   616→            <div>
   617→              <Label htmlFor="amount" className="text-sm font-medium">Montant (FCFA)</Label>

   620→                type="number"

   631→            <div className="flex gap-2">

   647→          </div>
   648→
   649→          <DialogFooter className="gap-2 sm:gap-0">
   650→            <DialogClose asChild>

   653→            <Button
   654→              className="flex-1 sm:flex-none bg-gradient-to-r from-orange-500 to-amber-500 hover:from-orange-600 hover:to-amber-600 text-white"
   655→              onClick={handleRecharge}

   659→            </Button>

   661→        </DialogContent>
   662→      </Dialog>
   663→
   664→      {/* Desktop Footer */}
