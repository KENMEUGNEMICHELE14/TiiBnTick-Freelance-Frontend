   136→// Deterministic number formatter (avoids hydration mismatch between server and client)
   137→function fmt(n: number | undefined): string {
   138→  if (n == null) return '0'
   139→  return n.toString().replace(/\B(?=(\d{3})+(?!\d))/g, '\u00A0')
